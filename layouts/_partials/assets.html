{{- /* The asset pipeline. Returns a dict of published URLs for one page in one locale: css site-wide stylesheet bundle (locale-independent) boot render-blocking bundle: browser-support.js + theme.js core per-locale bundle: translations, shared chrome, index.js, vendored driver.js + twemoji page per-page bundle, ending in the bootstrap() call that starts main() Args (dict): loc (locale directory, e.g. "en_US"), slug (page slug; "index" for the home page). Callable for any locale from any render pass — that is what lets layouts/sw.js enumerate every locale's bundles. Every third-party dependency listed under params.vendor is fetched once at build time and concatenated in, so the running site issues no cross-origin requests at all. */ -}} {{- /* resources.Concat needs a typed resource.Resources slice, which `slice` does not produce — an empty resources.Match does, and append preserves it. */ -}} {{- $none := resources.Match "__no_such_asset__/*" -}} {{- $loc (resources.GetRemote $url) -}} {{- with $t.Err -}}{{- errorf "vendor %s (%s): %s" $k $url . -}}{{- end -}} {{- if not $t.Value -}}{{- errorf "vendor %s: fetch of %s returned nothing" $k $url -}}{{- end -}} {{- $vendor = merge $vendor (dict $k $t.Value) -}} {{- end -}} {{- /* ---------- self-hosted webfont ---------- */ -}} {{- $f := hugo.Data.font -}} {{- $faces := slice -}} {{- $fontUrls := slice -}} {{- range $f.subsets -}} {{- $url := printf "%s/ubuntu-%s-%d-normal.woff2" $f.base .name $f.weight -}} {{- $t := try (resources.GetRemote $url) -}} {{- with $t.Err -}}{{- errorf "font %s: %s" $url . -}}{{- end -}} {{- if not $t.Value -}}{{- errorf "font: fetch of %s returned nothing" $url -}}{{- end -}} {{- $pub := $t.Value | resources.Copy (printf "/fonts/ubuntu-%s-%d.woff2" .name $f.weight) | fingerprint -}} {{- $pub.Publish -}} {{- $fontUrls = $fontUrls | append $pub.RelPermalink -}} {{- $faces = $faces | append (printf "@font-face{font-family:'%s';font-style:normal;font-weight:%d;font-display:swap;src:url(%s) format('woff2');unicode-range:%s}" $f.family $f.weight $pub.RelPermalink .unicodeRange) -}} {{- end -}} {{- $fontCss := resources.FromString "styles/_font.css" (delimit $faces "") -}} {{- /* ---------- stylesheets ---------- */ -}} {{- /* Order is the cascade contract inherited from the old templates: the generated @font-face rules first, then the global components, then driver.css *before* tutorial.css so the theme-aware .tutorial-popover overrides still win, then the page stylesheets. */ -}} {{- $cssFiles := slice "styles/components/base.css" "styles/components/header.css" "styles/components/select-box.css" "styles/components/char-loading.css" "styles/components/card.css" $css | append $vendor.driver_css (resources.Get "styles/components/tutorial.css") -}} {{- range (slice "index" "deck" "deck-edit" "account" "marketplace" "privacy" "licenses") -}} {{- $r := resources.Get (printf "styles/pages/%s.css" .) -}} {{- if not $r -}}{{- errorf "missing page stylesheet %s" . -}}{{- end -}} {{- $css = $css | append $r -}} {{- end -}} {{- $cssBundle := $css | "scripts/index.js" "scripts/components/daily-streak.js") -}} {{- $r := resources.Get . -}} {{- if not $r -}}{{- errorf "missing core script %s" . -}}{{- end -}} {{- $core = $core | append $r -}} {{- end -}} {{- $core = $core | append $vendor.driver_js (resources.Get "scripts/components/tutorial.js") -}} {{- $coreBundle := $core | resources.Concat (printf "scripts/core.%s.js" $loc) | minify | fingerprint -}} {{- /* ---------- page bundle ---------- */ -}} {{- /* `pre` held the scripts the old templates put *before* the shared footer (they only define things main()'s async continuations
